= RACI y GAPs — evaluación por cliente

Herramienta interactiva para *facilitar el taller* con un cliente concreto. A partir del mapa teórico de xref:00-01-lab-a-prd.adoc[Lab → Prd], evalúa cada capacidad, reparte responsabilidades (RACI) y exporta la foto de gaps para entregarla.

* Todo corre en el navegador: el estado se guarda en `localStorage` y se exporta como fichero (*Markdown* para entregar, *JSON* para retomar después).
* El *RACI* parte del borrador del lab; ajústalo por cliente. La herramienta avisa de filas sin *A* (Accountable) o con *A* duplicado.
* El *resumen* se calcula solo: gaps, parciales, alertas de gobierno y carga por equipo.

  function slug(s) {

    return 'raci-gaps-' + slug(state.cliente) + '-' + (state.fecha || hoy()) + '.' + ext;
  }
  function download(contents, ext, mime) {

    var a = document.createElement('a');
    a.href = URL.createObjectURL(blob);
    a.download = filename(ext);
    document.body.appendChild(a);
    a.click();
    URL.revokeObjectURL(a.href);
    a.remove();
  }

  $('rg-load').addEventListener('change', function () {
    var file = this.files[0];
    this.value = '';
    if (!file) return;
    var reader = new FileReader();
    reader.onload = function () {
      try {
        var saved = JSON.parse(reader.result);
        if (!saved || !saved.areas) throw new Error('formato');
        state.cliente = saved.cliente || '';
        state.fecha = saved.fecha || hoy();
        state.facilitador = saved.facilitador || '';
        state.cargado = file.name;
        AREAS.forEach(function (a) {
          if (saved.areas[a.id]) state.areas[a.id] = Object.assign(state.areas[a.id], saved.areas[a.id]);
        });
        if (saved.raci) Object.keys(saved.raci).forEach(function (i) {
          if (state.raci[i]) Object.assign(state.raci[i], saved.raci[i]);
        });
        save();
        applyState();
        renderAll();
      } catch (e) {
        alert('JSON no válido: se esperaba un fichero exportado por esta herramienta.');
      }
    };
    reader.readAsText(file);
  });

      state.raci[i] = {};
      TEAMS.forEach(function (t) { state.raci[i][t] = act.raci[t]; });
    });
    applyState();
    renderAll();
  });
